# 停车位预约管理系统 - 完整开发方案

基于 Vibe Coding 开发模式，我为你设计了一套完整的企业园区停车位预约管理系统。采用 **Vue3 + Node.js + SQLite** 技术栈

------

## 一、系统架构总览

### 技术栈选型

表格







|  层级  |               技术选型                |               说明               |

## 二、数据库设计

### 数据库表结构（SQLite）

sql







```
-- 用户表
CREATE TABLE users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  username TEXT UNIQUE NOT NULL,
  password TEXT NOT NULL,
  real_name TEXT NOT NULL,
  department TEXT,
  phone TEXT,
  role TEXT NOT NULL DEFAULT 'employee', -- employee / admin / super_admin
  violation_count INTEGER DEFAULT 0,
  ban_until DATETIME,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 车辆表
CREATE TABLE vehicles (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL,
  plate_number TEXT UNIQUE NOT NULL,
  car_type TEXT,
  color TEXT,
  is_default INTEGER DEFAULT 0,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 车位表
CREATE TABLE parking_spots (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  spot_code TEXT UNIQUE NOT NULL, -- 车位编号 A001, B023
  area TEXT NOT NULL, -- 区域 A区/B区/C区
  spot_type TEXT NOT NULL, -- fixed/shared/visitor/charging
  status TEXT NOT NULL DEFAULT 'available', -- available/occupied/maintenance
  position_x INTEGER, -- 地图坐标
  position_y INTEGER,
  floor INTEGER DEFAULT 1
);

-- 预约表
CREATE TABLE reservations (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL,
  spot_id INTEGER NOT NULL,
  vehicle_id INTEGER NOT NULL,
  reserve_date DATE NOT NULL,
  time_slot TEXT NOT NULL, -- morning/afternoon/all_day
  status TEXT NOT NULL DEFAULT 'reserved', -- reserved/checked_in/cancelled/expired/violation
  checkin_time DATETIME,
  expire_time DATETIME,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id),
  FOREIGN KEY (spot_id) REFERENCES parking_spots(id),
  FOREIGN KEY (vehicle_id) REFERENCES vehicles(id)
);

-- 违约记录表
CREATE TABLE violation_records (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL,
  reservation_id INTEGER NOT NULL,
  reason TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id),
  FOREIGN KEY (reservation_id) REFERENCES reservations(id)
);
```

#### 2. 数据库初始化 `db.js`

javascript



运行





```
const sqlite3 = require('sqlite3').verbose();
const path = require('path');
const bcrypt = require('bcryptjs');

const db = new sqlite3.Database(path.join(__dirname, 'parking.db'));

// 初始化数据表
function initDB() {
  db.serialize(() => {
    // 创建表...（执行上面的SQL）
    
    // 初始化管理员账号 admin/123456
    const salt = bcrypt.genSaltSync(10);
    const hash = bcrypt.hashSync('123456', salt);
    
    db.run(`INSERT OR IGNORE INTO users (username, password, real_name, role) 
            VALUES ('admin', ?, '系统管理员', 'super_admin')`, [hash]);
    
    // 初始化200个车位数据
    initParkingSpots();
  });
}

function initParkingSpots() {
  const areas = ['A区', 'B区', 'C区', 'D区'];
  const types = ['fixed', 'shared', 'shared', 'shared', 'visitor', 'charging'];
  
  let stmt = db.prepare('INSERT INTO parking_spots (spot_code, area, spot_type, position_x, position_y) VALUES (?, ?, ?, ?, ?)');
  
  let spotId = 1;
  areas.forEach((area, areaIdx) => {
    for (let i = 1; i <= 50; i++) {
      const code = `${area.charAt(0)}${String(i).padStart(3, '0')}`;
      const type = types[Math.floor(Math.random() * types.length)];
      const x = (i % 10) * 60 + areaIdx * 200 + 50;
      const y = Math.floor(i / 10) * 40 + 50;
      stmt.run(code, area, type, x, y);
      spotId++;
    }
  });
  stmt.finalize();
}

module.exports = { db, initDB };
```

#### 3. JWT 鉴权中间件 `middleware/auth.js`

javascript



运行





```
const jwt = require('jsonwebtoken');
const JWT_SECRET = 'parking-system-secret-key-2024';

function authMiddleware(roles = []) {
  return (req, res, next) => {
    const token = req.headers.authorization?.split(' ')[1];
    if (!token) return res.status(401).json({ message: '未登录' });
    
    try {
      const decoded = jwt.verify(token, JWT_SECRET);
      req.user = decoded;
      
      if (roles.length && !roles.includes(decoded.role)) {
        return res.status(403).json({ message: '权限不足' });
      }
      next();
    } catch (e) {
      res.status(401).json({ message: 'Token无效' });
    }
  };
}

module.exports = { authMiddleware, JWT_SECRET };
```

#### 4. 预约核心逻辑 `routes/reservations.js`

javascript



运行





```
const express = require('express');
const router = express.Router();
const dayjs = require('dayjs');
const { db } = require('../db');
const { authMiddleware } = require('../middleware/auth');

// 创建预约
router.post('/', authMiddleware(), (req, res) => {
  const userId = req.user.id;
  const { spotId, vehicleId, reserveDate, timeSlot } = req.body;
  
  // 1. 检查是否被封禁
  db.get('SELECT ban_until, violation_count FROM users WHERE id = ?', [userId], (err, user) => {
    if (user.ban_until && dayjs(user.ban_until).isAfter(dayjs())) {
      return res.status(400).json({ message: '您处于违约封禁期，暂无法预约' });
    }
    
    // 2. 预约规则：普通员工提前3天
    const maxAdvanceDays = req.user.role === 'employee' ? 3 : 30;
    const targetDate = dayjs(reserveDate);
    if (targetDate.diff(dayjs(), 'day') > maxAdvanceDays) {
      return res.status(400).json({ message: `最多只能提前${maxAdvanceDays}天预约` });
    }
    
    // 3. 检查同一时段是否已有预约
    db.get(`SELECT id FROM reservations 
            WHERE user_id = ? AND reserve_date = ? AND time_slot = ? 
            AND status IN ('reserved', 'checked_in')`, 
      [userId, reserveDate, timeSlot], (err, existing) => {
      if (existing) return res.status(400).json({ message: '该时段您已有有效预约' });
      
      // 4. 检查车位是否被占用
      db.get(`SELECT id FROM reservations 
              WHERE spot_id = ? AND reserve_date = ? AND time_slot = ? 
              AND status IN ('reserved', 'checked_in')`,
        [spotId, reserveDate, timeSlot], (err, spotTaken) => {
        if (spotTaken) return res.status(400).json({ message: '该车位已被预约' });
        
        // 5. 创建预约
        const expireTime = dayjs(reserveDate).hour(9).minute(0).toISOString(); // 上午9点前未核销自动释放
        db.run(`INSERT INTO reservations (user_id, spot_id, vehicle_id, reserve_date, time_slot, expire_time)
                VALUES (?, ?, ?, ?, ?, ?)`,
          [userId, spotId, vehicleId, reserveDate, timeSlot, expireTime],
          function(err) {
            if (err) return res.status(500).json({ message: '预约失败' });
            res.json({ id: this.lastID, message: '预约成功' });
          });
      });
    });
  });
});

// 核销入场
router.post('/:id/checkin', authMiddleware(), (req, res) => {
  const reservationId = req.params.id;
  
  db.get('SELECT * FROM reservations WHERE id = ?', [reservationId], (err, record) => {
    if (!record) return res.status(404).json({ message: '预约不存在' });
    if (record.user_id !== req.user.id && req.user.role === 'employee') {
      return res.status(403).json({ message: '无权操作' });
    }
    
    db.run(`UPDATE reservations SET status = 'checked_in', checkin_time = CURRENT_TIMESTAMP WHERE id = ?`,
      [reservationId], (err) => {
      res.json({ message: '核销成功' });
    });
  });
});

module.exports = router;
```

#### 5. 定时任务（违约自动处理）`utils/scheduler.js`

javascript



运行





```
const cron = require('node-cron');
const { db } = require('../db');

// 每分钟检查超时未核销的预约
function startScheduler() {
  cron.schedule('* * * * *', () => {
    // 标记超时预约为违约
    db.all(`SELECT r.id, r.user_id FROM reservations r
            WHERE r.status = 'reserved' AND r.expire_time < DATETIME('now')`,
      (err, rows) => {
      rows.forEach(row => {
        db.run(`UPDATE reservations SET status = 'violation' WHERE id = ?`, [row.id]);
        
        // 记录违约
        db.run(`INSERT INTO violation_records (user_id, reservation_id, reason) VALUES (?, ?, '超时未核销')`,
          [row.user_id, row.id]);
        
        // 更新用户违约次数
        db.run(`UPDATE users SET violation_count = violation_count + 1 WHERE id = ?`, [row.user_id]);
        
        // 累计3次封禁1周
        db.get(`SELECT violation_count FROM users WHERE id = ?`, [row.user_id], (err, user) => {
          if (user.violation_count >= 3) {
            const banUntil = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000).toISOString();
            db.run(`UPDATE users SET ban_until = ?, violation_count = 0 WHERE id = ?`, [banUntil, row.user_id]);
          }
        });
      });
    });
  });
}

module.exports = { startScheduler };
```

## 五、核心 API 接口清单

### 认证模块

- `POST /api/auth/login` - 用户登录
- `POST /api/auth/logout` - 退出登录

- `GET /api/spots` - 获取车位列表（支持筛选）
- `GET /api/spots/map-status` - 获取地图状态（含预约情况）
- `POST /api/spots` - 新增车位（管理员）
- `PUT /api/spots/:id` - 更新车位信息
- `DELETE /api/spots/:id` - 删除车位

### 预约管理

- `POST /api/reservations` - 创建预约
- `GET /api/reservations/my` - 我的预约列表
- `POST /api/reservations/:id/checkin` - 核销入场
- `POST /api/reservations/:id/cancel` - 取消预约
- `GET /api/reservations` - 全部预约列表（管理员）

### 车辆管理

- `GET /api/vehicles` - 获取我的车辆
- `POST /api/vehicles` - 添加车辆
- `PUT /api/vehicles/:id` - 更新车辆
- `DELETE /api/vehicles/:id` - 删除车辆

### 统计报表

- `GET /api/stats/overview` - 概览统计

## 七、业务规则实现要点

1. **预约时间窗口**：普通员工提前 3 天，管理员可配置
2. **一人一时段一位**：同一日期时段只能有一个有效预约
3. **自动释放机制**：预约时段开始后 15 分钟未核销，自动释放并记违约
4. **违约累计规则**：累计 3 次违约封禁 7 天，封禁期满自动清零
5. **车位类型限制**：固定车位不可预约，访客车位需管理员授权
6. **时段划分**：上午（8:00-12:00）、下午（13:00-18:00）、全天
